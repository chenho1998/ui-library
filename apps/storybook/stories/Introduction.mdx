## Quick usage

```tsx
import '@yourorg/ui/styles.css';
import { ThemeProvider, Toaster, Button } from '@yourorg/ui';

export function App() {
  return (
    <ThemeProvider defaultColorScheme="system" storageKey="color-scheme">
      <Button>Hello</Button>
      <Toaster />
    </ThemeProvider>
  );
}
```

## Custom theming

```tsx
const acme = createTheme({
  brand: '#7c3aed',                         // generates the full OKLCH brand palette
  tokens: { radius: { md: '6px' } },        // global token overrides
  colors: { dark: { bg: '#0b0b10' } },      // per-scheme semantic overrides
  components: { 'button-radius': '999px' }, // component tokens
});

<ThemeProvider theme={acme}>…</ThemeProvider>
```

Use the **Scheme**, **Theme** and **Font** toolbar buttons above to test light/dark, built-in expressive themes and
OpenDyslexic on any page.
